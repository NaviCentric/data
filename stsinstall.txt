param(
    [Parameter(Mandatory = $true)]
    [string]$Version
)

$InstallDir  = "C:\Jenkins\workspace\Test_Automation\demo_desktop\test-artifacts\desktop"
$FileName    = "STS-PluginAndDesktop_v$Version.exe"
$InstallerPath = Join-Path $InstallDir $FileName
$LogFile     = "D:\Temp\stsinstall.log"

function Write-Log {
    param([string]$Message)
    $timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    "$timestamp - $Message" | Out-File $LogFile -Append
}

Write-Log "---- stsinstall started (Version=$Version) ----"

if (-not (Test-Path $InstallerPath)) {
    Write-Log "ERROR: Installer not found at '$InstallerPath'"
    exit 1
}

Write-Log "Running installer: $InstallerPath /S"

try {
    $process = Start-Process -FilePath $InstallerPath -ArgumentList "/S" -Wait -PassThru
    $exitCode = $process.ExitCode
}
catch {
    Write-Log "ERROR: Failed to launch installer - $($_.Exception.Message)"
    exit 1
}

if ($exitCode -eq 0) {
    Write-Log "Install succeeded (Version=$Version, ExitCode=$exitCode)"
    exit 0
} else {
    Write-Log "ERROR: Install failed (Version=$Version, ExitCode=$exitCode)"
    exit $exitCode
}
